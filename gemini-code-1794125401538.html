💬 ความคิดเห็นทั้งหมด (ถาวร)

    📜 การเชื่อมต่อ Google Apps Script

    วาง Web App URL ของคุณที่นี่เพื่อบันทึกข้อมูลลง Google Sheets ถาวรข้ามเครื่อง:

    คัดลอกโค้ด Google Apps Script นี้ไปวางใน Google Sheets:

    function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet();
  var scores = sheet.getSheetByName("Scores").getDataRange().getValues();
  var comments = sheet.getSheetByName("Comments").getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify({
    scores: scores.slice(1),
    comments: comments.slice(1)
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp.getActiveSpreadsheet();
  if (data.action === "saveScore") {
    var sSheet = sheet.getSheetByName("Scores") || sheet.insertSheet("Scores");
    if(sSheet.getLastRow() === 0) sSheet.appendRow(["Name", "Score", "Date"]);
    sSheet.appendRow([data.name, data.score, new Date()]);
  } else if (data.action === "saveComment") {
    var cSheet = sheet.getSheetByName("Comments") || sheet.insertSheet("Comments");
    if(cSheet.getLastRow() === 0) cSheet.appendRow(["Name", "Comment", "Date"]);
    cSheet.appendRow([data.name, data.comment, new Date()]);
  }
  return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
}
